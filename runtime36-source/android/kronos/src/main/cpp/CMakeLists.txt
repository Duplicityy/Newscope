cmake_minimum_required(VERSION 3.22.1)
project(kronosmobile)
add_library(kronosmobile SHARED kronos_jvm.cpp mobile_hd.cpp)
if(NOT DEFINED NEWSCAPE_OWNER_HOST OR NEWSCAPE_OWNER_HOST STREQUAL "")
    message(FATAL_ERROR "NEWSCAPE_OWNER_HOST must match the Android owner endpoint")
endif()
target_compile_definitions(kronosmobile PRIVATE NEWSCAPE_OWNER_HOST="${NEWSCAPE_OWNER_HOST}")
target_compile_features(kronosmobile PRIVATE cxx_std_17)
target_compile_options(kronosmobile PRIVATE -O2)
target_link_libraries(kronosmobile android log dl EGL GLESv3)
target_link_options(kronosmobile PRIVATE "-Wl,-z,max-page-size=16384")
